# 远景 / 洲视图生物群 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 远景按丰度散布的微缩模型（实例化 + 着色器待机动作），洲视图最近 ≤300 个个体升为完整 Rig 并有漫游 / 进食 / 睡觉 / 成群 / 追逐逃跑。

**Architecture:** 纯 TS 的散布（`simulation/scatter.ts`）、行为（`simulation/herdSim.ts`）与升档选择（`simulation/promotion.ts`）；渲染分两档：`scene/Living/Herds/Herds.tsx`（每物种 2 个 `InstancedMesh`，网点面 + 轮廓，尺寸 / 背面 / 待机动作在顶点着色器中）和 `HerdAgents.tsx`（Rig 对象池 + 行为）。两档通过模块级 `herdRuntime` 共享逐帧状态。

**Tech Stack:** React 19 · R3F 9 · three 0.186 · Zustand 5 · Vitest 5（jsdom）· Playwright

**设计稿：** `docs/superpowers/specs/2026-11-07-herds-design.md`。实现时相对设计稿的细化（已向用户说明）：
- 生境检查只在点位本身符合生境时才对周围个体生效；每点第 0 个个体总在点位上。
- 食性存于 `data/species/diet.ts`，在 `SPECIES` 导出时合并为 `Species.diet` 并在 `evidenceNote` 末尾追加"食性：…"。
- 古人类（`hominin` 原型）不参与追逐（双方都不）。
- 实例档点击用屏幕空间最近点判定（着色器缩放使射线检测失效）；完整档仍用 R3F `onClick`。
- 只有动物（`locomotion ≠ "none"`）升为完整档；植物在实例档已有同样的摇摆。
- 行为写在新的 `herdSim.ts`（角度单位）；`BehaviorSystem.ts` 的米制 `stepAgent` 留给观察窗，不改。

```ts
/** 两点间的大圆角距离（度）。 */
export function angularDistanceDeg(a: LonLat, b: LonLat): number {
  const c = Math.sin(a[1] * DEG) * Math.sin(b[1] * DEG) + Math.cos(a[1] * DEG) * Math.cos(b[1] * DEG) * Math.cos((a[0] - b[0]) * DEG);
  return Math.acos(Math.min(1, Math.max(-1, c))) / DEG;
}

/** 从 c 出发沿方位角 bearing（弧度，0 = 北，π/2 = 东）走 distDeg 度后的位置。 */
export function offsetLonLat(c: LonLat, bearing: number, distDeg: number): LonLat {
  if (distDeg === 0) return [c[0], c[1]];
  const p1 = c[1] * DEG;
  const d = distDeg * DEG;
  const p2 = Math.asin(Math.sin(p1) * Math.cos(d) + Math.cos(p1) * Math.sin(d) * Math.cos(bearing));
  const l2 = c[0] * DEG + Math.atan2(Math.sin(bearing) * Math.sin(d) * Math.cos(p1), Math.cos(d) - Math.sin(p1) * Math.sin(p2));
  return [((((l2 / DEG + 180) % 360) + 360) % 360) - 180, p2 / DEG];
}

/** 相机高度（地球半径）为 altitude 时，视野中心到地平线的角距离（度）。 */
export function horizonDeg(altitude: number): number {
  return Math.acos(1 / (1 + Math.max(0, altitude))) / DEG;
}
```

```ts
export function hashString(s: string): number {
  let h = 0x811c9dc5;
  for (let i = 0; i < s.length; i++) {
    h ^= s.charCodeAt(i);
    h = Math.imul(h, 0x01000193);
  }
  return h >>> 0;
```

```ts
import type { LonLat } from "../data/schema";
import { rasterizePolygons, type LandGrid } from "../utils/dissolve";
import { unwrapRing } from "../utils/geo";

export type { LandGrid };

/**
 * 当前切片轮廓 → 粗陆地网格（CPU 侧的海陆判定，散布时用）。
 * 与 drawLandMask 同样的规则：经度展开后在 0、±360° 各放一份；极地环补到极点闭合。
 * 每个环当作独立多边形（并集），湖泊等内环会被填成陆地——对散布无影响。
 */
export function landGridFromOutlines(outlines: LonLat[][], resDeg = 1): LandGrid {
  const polys: LonLat[][][] = [];
  for (const raw of outlines) {
    const ring = unwrapRing(raw);
    if (ring.length < 3) continue;
    const first = ring[0]!;
    const last = ring[ring.length - 1]!;
    let closed = ring;
    if (Math.abs(last[0] - first[0]) >= 359) {
      let minLat = Infinity;
      let maxLat = -Infinity;
      for (const p of ring) {
        if (p[1] < minLat) minLat = p[1];
        if (p[1] > maxLat) maxLat = p[1];
      }
      const pole = Math.abs(minLat) >= Math.abs(maxLat) ? -90 : 90;
      closed = [...ring, [last[0], pole], [first[0], pole]];
    }
    for (const off of [0, -360, 360]) polys.push([closed.map(([lon, lat]) => [lon + off, lat] as LonLat)]);
  }
  return rasterizePolygons(polys, resDeg);
}

export function isLandAt(g: LandGrid, [lon, lat]: LonLat): boolean {
  const x = ((((lon + 180) % 360) + 360) % 360) / g.resDeg;
  const i = Math.min(g.width - 1, Math.floor(x));
  const j = Math.min(g.height - 1, Math.max(0, Math.floor((90 - lat) / g.resDeg)));
  return g.cells[j * g.width + i] === 1;
}
```

```ts
import type { AnimationClipName, Diet, HabitatId, Species, VisualizationAsset } from "../data/schema";
import { abundanceTier, type AbundanceTier } from "./abundance";

const LAND: HabitatId[] = ["terrestrial", "forest", "steppe", "wetland"];

/** 散布时的海陆要求：只有纯海洋生物必须在海里；淡水 / 滨海生物不检查（内陆水体在遮罩上是陆地）。 */
export type HabitatNeed = "land" | "water" | "any";

export function placementNeed(h: HabitatId[]): HabitatNeed {
  if (h.includes("coastal") || h.includes("freshwater")) return "any";
  const land = h.some((x) => LAND.includes(x));
  const sea = h.includes("marine");
  if (land && sea) return "any";
  return land ? "land" : sea ? "water" : "any";
}

/** 追逐配对用的生境组：陆生 / 水生，滨海两者都算。 */
export interface HabitatGroups {
  land: boolean;
  water: boolean;
}

export function habitatGroups(h: HabitatId[]): HabitatGroups {
  const coastal = h.includes("coastal");
  return { land: coastal || h.some((x) => LAND.includes(x)), water: coastal || h.includes("marine") || h.includes("freshwater") };
}

export function shareHabitat(a: HabitatGroups, b: HabitatGroups): boolean {
  return (a.land && b.land) || (a.water && b.water);
}

/** 丰度等级：使用 GBIF 现代分布显示的现生类群按 GBIF 记录数，其余按 PBDB 化石记录数。 */
export function speciesTier(pbdbOccurrences: number, gbifRecords: number | undefined): AbundanceTier {
  return gbifRecords !== undefined ? abundanceTier(gbifRecords, "gbif") : abundanceTier(pbdbOccurrences);
}

/** 行为系统需要的个体特征（与渲染无关）。clips 为该原型实际实现的动画片段。 */
export interface HerdTraits {
  speciesId: string;
  diet: Diet | undefined;
  sizeM: number;
  locomotion: VisualizationAsset["locomotion"];
  groups: HabitatGroups;
  canEat: boolean;
  canSleep: boolean;
  /** 古人类不参与追逐（避免不当暗示）。 */
  hominin: boolean;
  /** 植食 / 滤食：同一点位的个体成群移动。 */
  herds: boolean;
}

export function traitsFor(species: Species, asset: VisualizationAsset, clips: AnimationClipName[]): HerdTraits {
  return {
    speciesId: species.id,
    diet: species.diet,
    sizeM: asset.sizeM,
    locomotion: asset.locomotion,
    groups: habitatGroups(species.habitatIds),
    canEat: clips.includes("eat") && asset.animationSet.includes("eat"),
    canSleep: clips.includes("sleep") && asset.locomotion !== "fly" && asset.locomotion !== "none",
    hominin: asset.archetype === "hominin",
    herds: asset.locomotion !== "none" && (species.diet === "herbivore" || species.diet === "filter"),
  };
}
```

```ts
import type { LonLat } from "../data/schema";
import { angularDistanceDeg, horizonDeg, offsetLonLat } from "../utils/geo";
import type { AbundanceTier } from "./abundance";
import { seededRandom } from "./BehaviorSystem";
import { hashString } from "./hash";
import type { HabitatNeed } from "./herdTraits";
import { isLandAt, type LandGrid } from "./landGrid";

/**
 * 按丰度散布的生物群（设计：docs/superpowers/specs/2026-11-07-herds-design.md §2）。纯函数。
 * 个体挂在"物种 + 点下标 + 序号"上，偏移（方位角 + 角距离）固定，随点位漂移整体移动。
 */

/** 相机高度（地球半径）低于该值时显示生物群（实例档）。 */
export const HERD_MAX_ALTITUDE = 4;
/** 低于该高度为洲视图：植物更茂密，动物可升为完整档。 */
export const CONTINENT_ALTITUDE = 1.2;

const FAR_COUNT: Record<AbundanceTier, number> = { 1: 1, 2: 2, 3: 4, 4: 8 };

export function individualsPerSite(tier: AbundanceTier, stationary: boolean, continental: boolean): number {
  return FAR_COUNT[tier] * (continental && stationary ? 3 : 1);
}

/** 散布半径（度）：0.3°–1.2°，按丰度等级线性增大。 */
export function scatterRadiusDeg(tier: AbundanceTier): number {
  return 0.3 * tier;
}

export interface HerdSpeciesInput {
  speciesId: string;
  /** 当前参考系下可重建的点位。 */
  sites: LonLat[];
  /** 与 sites 一一对应：点在原始点列表（化石点或 GBIF 点）中的下标。 */
  siteIndex: number[];
  perSite: number;
  radiusDeg: number;
  need: HabitatNeed;
}

export interface HerdIndividual {
  key: string;
  speciesId: string;
  siteIndex: number;
  k: number;
  /** 相对点位的方位角（弧度）与角距离（度）。 */
  bearing: number;
  distDeg: number;
  /** 模型朝向（弧度，从北向东）与动画相位。 */
  heading: number;
  phase: number;
}

const fits = (g: LandGrid, p: LonLat, need: HabitatNeed) => need === "any" || isLandAt(g, p) === (need === "land");

function scatterSite(s: HerdSpeciesInput, i: number, grid: LandGrid | null): HerdIndividual[] {
  const siteIndex = s.siteIndex[i]!;
  const center = s.sites[i]!;
  const rand = seededRandom(hashString(`${s.speciesId}@${siteIndex}`));
  // 点位本身不符合生境（粗分辨率海岸线下常见）时不检查周围个体
  const check = grid !== null && s.need !== "any" && fits(grid, center, s.need);
  const out: HerdIndividual[] = [];
  for (let k = 0; k < s.perSite; k++) {
    let bearing = 0;
    let distDeg = 0;
    let ok = k === 0;
    for (let t = 0; !ok && t < 3; t++) {
      bearing = rand() * Math.PI * 2;
      distDeg = Math.sqrt(rand()) * s.radiusDeg;
      ok = !check || fits(grid!, offsetLonLat(center, bearing, distDeg), s.need);
    }
    const heading = rand() * Math.PI * 2;
    const phase = rand() * 10;
    if (ok) out.push({ key: `${s.speciesId}#${siteIndex}#${k}`, speciesId: s.speciesId, siteIndex, k, bearing, distDeg, heading, phase });
  }
  return out;
}

/**
 * 选点并散布：只取地平线以内的点；每个物种先保留哈希排名最靠前的一个点，
 * 其余点按固定哈希排名依次加入，直到个体总数达到 cap。排名与视角无关，相机小幅移动时选点稳定。
 */
export function buildHerd(species: HerdSpeciesInput[], view: { center: LonLat; altitude: number }, cap: number, grid: LandGrid | null): HerdIndividual[] {
  const limit = horizonDeg(view.altitude) + 3;
  type C = { s: HerdSpeciesInput; i: number; rank: number };
  const firsts: C[] = [];
  const rest: C[] = [];
  for (const s of species) {
    let best: C | null = null;
    for (let i = 0; i < s.sites.length; i++) {
      if (angularDistanceDeg(view.center, s.sites[i]!) > limit) continue;
      const c: C = { s, i, rank: hashString(`${s.speciesId}#${s.siteIndex[i]}`) };
      if (!best || c.rank < best.rank) {
        if (best) rest.push(best);
        best = c;
      } else rest.push(c);
    }
    if (best) firsts.push(best);
  }
  rest.sort((a, b) => a.rank - b.rank);
  const out: HerdIndividual[] = [];
  for (const c of firsts) out.push(...scatterSite(c.s, c.i, grid));
  for (const c of rest) if (out.length + c.s.perSite <= cap) out.push(...scatterSite(c.s, c.i, grid));
  return out;
}
```

```ts
import type { AnimationClipName, LonLat, VisualizationAsset } from "../data/schema";
import { seededRandom } from "./BehaviorSystem";
import { hashString } from "./hash";
import { shareHabitat, type HerdTraits } from "./herdTraits";

/**
 * 地球上生物群的行为（设计 §4）：idle / wander / eat / sleep / chase / flee。
 * 地球上的模型不是真实比例，距离一律用角度（度）；时间为渲染时钟（秒），与模拟倍率无关。
 * 纯函数 + 可注入随机数。不表现捕获 / 死亡。
 */

export type HerdState = "idle" | "wander" | "eat" | "sleep" | "chase" | "flee";

export const WANDER_RADIUS_DEG: Record<VisualizationAsset["locomotion"], number> = { none: 0, crawl: 0.05, walk: 0.15, swim: 0.2, fly: 0.35 };
export const MAX_CHASES = 8;
const CHASE_RANGE_DEG = 1;
const CHASE_ESCAPE_DEG = 1.5;
/** 每只空闲的捕食者每秒尝试发起追逐的概率。 */
const CHASE_RATE = 0.2;
const HERD_WINDOW_S = 10;
const D = Math.PI / 180;

export interface HerdAgent {
  key: string;
  traits: HerdTraits;
  /** 家 = 散布位置（随板块漂移由渲染层每帧更新）。 */
  home: LonLat;
  /** 成群的分组（物种 + 点位）。 */
  group: string;
  /** 相对家的偏移（度，向东 / 向北）。 */
  e: number;
  n: number;
  /** 朝向（弧度，0 = 北，π/2 = 东）。 */
  heading: number;
  state: HerdState;
  timer: number;
  targetE: number;
  targetN: number;
  cooldown: number;
  partner: string | null;
  phase: number;
  /** 动画时间（按片段播放速率累积）。 */
  animT: number;
  /** 成群时相对共享目标的固定偏移。 */
  slotE: number;
  slotN: number;
}

export function createHerdAgent(key: string, traits: HerdTraits, home: LonLat, group: string, rand: () => number): HerdAgent {
  const r = WANDER_RADIUS_DEG[traits.locomotion];
  const a = rand() * Math.PI * 2;
  const d = Math.sqrt(rand()) * r * 0.5;
  return {
    key, traits, home, group,
    e: Math.cos(a) * d,
    n: Math.sin(a) * d,
    heading: rand() * Math.PI * 2,
    state: "idle",
    timer: 1 + rand() * 3,
    targetE: 0,
    targetN: 0,
    cooldown: 5 + rand() * 10,
    partner: null,
    phase: rand() * 10,
    animT: 0,
    slotE: (rand() - 0.5) * r * 0.4,
    slotN: (rand() - 0.5) * r * 0.4,
  };
}

export function agentLonLat(a: HerdAgent): LonLat {
  const c = Math.max(0.05, Math.cos(a.home[1] * D));
  return [a.home[0] + a.e / c, Math.max(-89.9, Math.min(89.9, a.home[1] + a.n))];
}

/** a → b 的相对位置（a 处的局部度，向东 / 向北）。 */
function delta(a: HerdAgent, b: HerdAgent): [number, number] {
  const pa = agentLonLat(a);
  const pb = agentLonLat(b);
  const dl = ((((pb[0] - pa[0] + 180) % 360) + 360) % 360) - 180;
  return [dl * Math.cos(pa[1] * D), pb[1] - pa[1]];
}

export function canPrey(pred: HerdTraits, prey: HerdTraits): boolean {
  if (pred.speciesId === prey.speciesId || pred.hominin || prey.hominin) return false;
  if (pred.diet !== "carnivore" && pred.diet !== "omnivore") return false;
  if (pred.locomotion === "none" || prey.locomotion === "none") return false;
  if (!prey.diet || prey.diet === "unknown") return false;
  if (!shareHabitat(pred.groups, prey.groups)) return false;
  const r = prey.sizeM / pred.sizeM;
  return r >= 0.1 && r <= 1.5;
}

/** 成群物种的共享漫游目标：同一分组在同一个 10 秒窗口内相同。 */
export function groupTarget(group: string, clock: number, radius: number): [number, number] {
  const r = seededRandom(hashString(`${group}#${Math.floor(clock / HERD_WINDOW_S)}`));
  const a = r() * Math.PI * 2;
  const d = Math.sqrt(r()) * radius;
  return [Math.cos(a) * d, Math.sin(a) * d];
}

export function herdClip(state: HerdState): { clip: AnimationClipName; rate: number } {
  if (state === "chase" || state === "flee") return { clip: "move", rate: 1.8 };
  if (state === "wander") return { clip: "move", rate: 1 };
  return { clip: state, rate: 1 };
}

const busy = (a: HerdAgent) => a.state === "chase" || a.state === "flee";

function pickWander(a: HerdAgent, clock: number, rand: () => number) {
  const r = WANDER_RADIUS_DEG[a.traits.locomotion];
  if (a.traits.herds) {
    const [e, n] = groupTarget(a.group, clock, r);
    a.targetE = e + a.slotE;
    a.targetN = n + a.slotN;
    return;
  }
  const ang = rand() * Math.PI * 2;
  const d = Math.sqrt(rand()) * r;
  a.targetE = Math.cos(ang) * d;
  a.targetN = Math.sin(ang) * d;
}

function steer(a: HerdAgent, de: number, dn: number, speed: number, dt: number, turn: number) {
  let d = Math.atan2(de, dn) - a.heading;
  d = Math.atan2(Math.sin(d), Math.cos(d));
  a.heading += Math.max(-turn * dt, Math.min(turn * dt, d));
  const step = speed * dt;
  a.e += Math.sin(a.heading) * step;
  a.n += Math.cos(a.heading) * step;
}

function clampHome(a: HerdAgent, max: number) {
  const r = Math.hypot(a.e, a.n);
  if (r > max) {
    a.e *= max / r;
    a.n *= max / r;
  }
}

function endChase(a: HerdAgent, rand: () => number) {
  const wasChasing = a.state === "chase";
  a.partner = null;
  a.state = wasChasing && a.traits.canEat && rand() < 0.5 ? "eat" : "idle";
  a.timer = 2 + rand() * 3;
  a.cooldown = 20 + rand() * 20;
}

function startChases(agents: HerdAgent[], dt: number, rand: () => number) {
  let active = agents.filter((a) => a.state === "chase").length;
  for (const pred of agents) {
    if (active >= MAX_CHASES) return;
    if (busy(pred) || pred.state === "sleep" || pred.cooldown > 0) continue;
    if (rand() >= dt * CHASE_RATE) continue;
    let best: HerdAgent | null = null;
    let bestD = CHASE_RANGE_DEG;
    for (const prey of agents) {
      if (prey === pred || busy(prey) || prey.cooldown > 0 || !canPrey(pred.traits, prey.traits)) continue;
      const [de, dn] = delta(pred, prey);
      const d = Math.hypot(de, dn);
      if (d < bestD) {
        best = prey;
        bestD = d;
      }
    }
    if (!best) continue;
    const dur = 4 + rand() * 4;
    pred.state = "chase";
    pred.timer = dur;
    pred.partner = best.key;
    best.state = "flee";
    best.timer = dur;
    best.partner = pred.key;
    active++;
  }
}

function stepOne(a: HerdAgent, dt: number, clock: number, rand: () => number, byKey: Map<string, HerdAgent>) {
  const t = a.traits;
  if (t.locomotion === "none") {
    a.state = "idle";
    return;
  }
  a.cooldown = Math.max(0, a.cooldown - dt);
  a.timer -= dt;
  const r = WANDER_RADIUS_DEG[t.locomotion];
  const v = r / 6;
  if (busy(a)) {
    const other = a.partner ? byKey.get(a.partner) : undefined;
    const [de, dn] = other ? delta(a, other) : [0, 0];
    if (!other || other.partner !== a.key || a.timer <= 0 || Math.hypot(de, dn) > CHASE_ESCAPE_DEG) {
      endChase(a, rand);
      if (other && other.partner === a.key) endChase(other, rand);
      return;
    }
    if (a.state === "chase") steer(a, de, dn, v * 1.6, dt, 4);
    else steer(a, -de, -dn, v * 1.4, dt, 4);
    clampHome(a, r * 3);
    return;
  }
  if (a.state === "wander") {
    const de = a.targetE - a.e;
    const dn = a.targetN - a.n;
    if (Math.hypot(de, dn) <= v * dt || a.timer <= 0) {
      if (t.locomotion === "fly") {
        pickWander(a, clock, rand);
        a.timer = 6 + rand() * 6;
      } else {
        a.state = t.canEat && rand() < 0.45 ? "eat" : "idle";
        a.timer = 2 + rand() * 3;
      }
      return;
    }
    steer(a, de, dn, v, dt, 2);
    clampHome(a, r);
    return;
  }
  if (a.timer > 0) return;
  if (a.state !== "sleep" && t.canSleep && rand() < 0.15) {
    a.state = "sleep";
    a.timer = 8 + rand() * 7;
    return;
  }
  a.state = "wander";
  pickWander(a, clock, rand);
  a.timer = 4 + rand() * 6;
}

/** 推进全部个体一步。dt 被限制在 0.25 s 以内；关闭行为时全部原地 idle。 */
export function stepHerd(agents: HerdAgent[], dtRaw: number, clock: number, enabled: boolean, rand: () => number): void {
  const dt = Math.min(Math.max(dtRaw, 0), 0.25);
  if (!enabled) {
    for (const a of agents) {
      a.state = "idle";
      a.partner = null;
    }
    return;
  }
  const byKey = new Map(agents.map((a) => [a.key, a]));
  startChases(agents, dt, rand);
  for (const a of agents) stepOne(a, dt, clock, rand, byKey);
}
```

```ts
/**
 * 完整档选择：sorted 为按离视野中心距离排好的候选键。
 * 已升档且排名在 keepWithin 以内的保留，其余名额按距离补足，总数不超过 max。
 * 进 max / 出 keepWithin 的滞回避免镜头微动时个体反复切换。
 */
export function choosePromoted(sorted: string[], current: ReadonlySet<string>, max: number, keepWithin: number): Set<string> {
  const out = new Set<string>();
  for (const k of sorted.slice(0, keepWithin)) {
    if (out.size >= max) break;
    if (current.has(k)) out.add(k);
  }
  for (const k of sorted) {
    if (out.size >= max) break;
    out.add(k);
  }
  return out;
}
```

```ts
import { Matrix4, Vector3, type Quaternion } from "three";
import type { LonLat } from "../../../data/schema";

const D = Math.PI / 180;
const m = new Matrix4();
const up = new Vector3();
const east = new Vector3();
const north = new Vector3();
const fwd = new Vector3();
const side = new Vector3();

/**
 * 地表点的模型朝向：局部 +Y → 地表法线，局部 +Z → heading 方向（弧度，0 = 北，π/2 = 东）。
 * 与 herdSim 的 heading / 东北偏移约定一致。东、北向量解析求出，极点附近也不退化。
 */
export function surfaceQuaternion([lon, lat]: LonLat, heading: number, out: Quaternion): Quaternion {
  const l = lon * D;
  const p = lat * D;
  up.set(Math.cos(p) * Math.sin(l), Math.sin(p), Math.cos(p) * Math.cos(l));
  east.set(Math.cos(l), 0, -Math.sin(l));
  north.set(-Math.sin(p) * Math.sin(l), Math.cos(p), -Math.sin(p) * Math.cos(l));
  fwd.copy(east).multiplyScalar(Math.sin(heading)).addScaledVector(north, Math.cos(heading));
  side.crossVectors(up, fwd);
  m.makeBasis(side, up, fwd);
  return out.setFromRotationMatrix(m);
}
```

```ts
import { BufferAttribute, Mesh, Vector3, type BufferGeometry, type Object3D } from "three";
import { mergeGeometries } from "three/examples/jsm/utils/BufferGeometryUtils.js";
import type { VisualizationAsset } from "../../../data/schema";
import { buildRig } from "../../Models/builders";
import type { Style } from "../../Models/rig";

/**
 * 实例档的模型：LOD1 Rig 的静止姿态合并为一个几何体（设计 §3）。
 * 附加属性：aOutDir（轮廓外扩方向，从部件中心指向顶点，与 Rig 的部件级轮廓一致）、
 * aPart（0 主体 / 1 头颈 / 2 尾 / 3 肢翼鳍，着色器待机动作用）、aH（顶点高度 / 体高）。
 * 折边线在 6–12 px 下只是噪点，不烘焙。
 */
export interface BakedSpecies {
  fill: BufferGeometry;
  extent: number;
  height: number;
  /** plain 风格没有轮廓。 */
  outline: boolean;
  dispose(): void;
}

export function partTag(joint: string): 0 | 1 | 2 | 3 {
  if (/^(head|neck|jaw|trunk)/.test(joint)) return 1;
  if (/^tail/.test(joint)) return 2;
  if (/^(leg|arm|wing|flipper|fin|pectoral|flap|tentacle|antenna|grasp)/.test(joint)) return 3;
  return 0;
}

function jointOf(o: Object3D, joints: Record<string, Object3D>): string {
  for (let p: Object3D | null = o.parent; p; p = p.parent) if (p.name && joints[p.name] === p) return p.name;
  return "";
}

export function bakeSpecies(asset: VisualizationAsset, style: Style): BakedSpecies {
  const rig = buildRig(asset, 1, style);
  rig.root.updateMatrixWorld(true);
  const pieces: BufferGeometry[] = [];
  const v = new Vector3();
  const center = new Vector3();
  rig.root.traverse((o) => {
    if (!(o instanceof Mesh) || o.name === "outline") return;
    const g = o.geometry.index ? o.geometry.toNonIndexed() : o.geometry.clone();
    for (const name of Object.keys(g.attributes)) if (name !== "position" && name !== "normal") g.deleteAttribute(name);
    g.applyMatrix4(o.matrixWorld);
    const pos = g.getAttribute("position");
    const n = pos.count;
    center.setFromMatrixPosition(o.matrixWorld);
    const out = new Float32Array(n * 3);
    for (let i = 0; i < n; i++) {
      v.fromBufferAttribute(pos, i).sub(center);
      if (v.lengthSq() < 1e-12) v.set(0, 1, 0);
      v.normalize().toArray(out, i * 3);
    }
    g.setAttribute("aOutDir", new BufferAttribute(out, 3));
    g.setAttribute("aPart", new BufferAttribute(new Float32Array(n).fill(partTag(jointOf(o, rig.joints))), 1));
    pieces.push(g);
  });
  const fill = mergeGeometries(pieces);
  pieces.forEach((p) => p.dispose());
  if (!fill) throw new Error(`bakeSpecies: cannot merge ${asset.id}`);
  const pos = fill.getAttribute("position");
  const h = new Float32Array(pos.count);
  for (let i = 0; i < pos.count; i++) h[i] = Math.max(0, pos.getY(i)) / rig.height;
  fill.setAttribute("aH", new BufferAttribute(h, 1));
  const { extent, height } = rig;
  rig.dispose();
  return { fill, extent, height, outline: style !== "plain", dispose: () => fill.dispose() };
}
```

```ts
import { BackSide, MeshBasicMaterial, type MeshLambertMaterial, type WebGLProgramParametersWithUniforms } from "three";
import type { VisualizationAsset } from "../../../data/schema";
import { INK } from "../../Ink/inkPalette";
import { createToneMaterial, TONE_SHARED } from "../../Ink/toneMaterial";
import type { Style } from "../../Models/rig";

/**
 * 实例档材质（设计 §3）：在网点 / 轮廓材质上加入
 *   - 待机动作（顶点着色器，按 aPart / aH，时钟只随真实时间）；
 *   - 屏幕恒定尺寸（uPx CSS 像素 → 世界尺寸，按实例中心到相机的距离）；
 *   - 背面隐藏与升档隐藏（aHerd.y = 0）：缩放为 0。
 * 场景没有灯光，网点自算光照（沿用 createToneMaterial）。
 */

/** 全部生物群共享：动画时钟（秒）与设备像素比（Herds 每帧写入）。 */
export const HERD_SHARED = { uHerdTime: { value: 0 }, uDpr: { value: 1 } };

/** 每物种一份：屏幕尺寸（CSS px）、模型外形最大尺寸、动作类型。 */
export interface HerdUniforms {
  uPx: { value: number };
  uExtent: { value: number };
  uMotion: { value: number };
}

export function createHerdUniforms(extent: number, motion: number): HerdUniforms {
  return { uPx: { value: 12 }, uExtent: { value: extent }, uMotion: { value: motion } };
}

/** 0 植物 / 固着：弯曲摇摆；1 陆生：起伏 + 转头；2 游泳：摆尾；3 飞行：上下浮动。 */
export function motionKind(asset: VisualizationAsset): number {
  switch (asset.locomotion) {
    case "none":
      return 0;
    case "swim":
      return 2;
    case "fly":
      return 3;
    default:
      return 1;
  }
}

const VERT_PARS = /* glsl */ `
uniform float uHerdTime;
uniform float uDpr;
uniform float uPx;
uniform float uExtent;
uniform float uMotion;
uniform float uViewportH;
attribute vec2 aHerd;
attribute float aPart;
attribute float aH;
`;

const VERT_BODY = /* glsl */ `
#ifdef USE_INSTANCING
vec4 herdC = modelMatrix * instanceMatrix * vec4(0.0, 0.0, 0.0, 1.0);
#else
vec4 herdC = modelMatrix * vec4(0.0, 0.0, 0.0, 1.0);
#endif
float herdT = uHerdTime + aHerd.x;
if (uMotion < 0.5) {
  float bend = aH * aH;
  transformed.x += 0.06 * bend * sin(herdT * 1.2);
  transformed.z += 0.04 * bend * sin(herdT * 0.9 + 1.3);
} else if (uMotion < 1.5) {
  transformed.y += 0.015 * sin(herdT * 1.6);
  if (aPart > 0.5 && aPart < 1.5) transformed.x += 0.04 * sin(herdT * 0.4) * step(0.6, sin(herdT * 0.13));
} else if (uMotion < 2.5) {
  float tail = (aPart > 1.5 && aPart < 2.5) ? 1.0 : 0.0;
  transformed.x += (0.02 + 0.08 * tail) * sin(herdT * 3.0 - transformed.z * 2.0);
} else {
  transformed.y += 0.05 * sin(herdT * 1.5);
}
vec3 herdN = normalize(herdC.xyz);
float herdVisible = step(0.05, dot(herdN, normalize(cameraPosition - herdC.xyz))) * aHerd.y;
float herdDist = length((viewMatrix * herdC).xyz);
float herdScale = uPx * uDpr * 2.0 * herdDist / (projectionMatrix[1][1] * uViewportH) / uExtent;
transformed *= herdScale * herdVisible;
`;

function inject(shader: WebGLProgramParametersWithUniforms, u: HerdUniforms, extraPars = "") {
  Object.assign(shader.uniforms, HERD_SHARED, { uViewportH: TONE_SHARED.uViewportH }, u);
  shader.vertexShader = shader.vertexShader
    .replace("#include <common>", `#include <common>\n${VERT_PARS}${extraPars}`)
    .replace("#include <project_vertex>", `${VERT_BODY}\n#include <project_vertex>`);
}

export function createHerdToneMaterial(style: Style, u: HerdUniforms): MeshLambertMaterial {
  const mat = createToneMaterial({ base: INK.sheet, mode: style, dots: false });
  const inner = mat.onBeforeCompile;
  mat.onBeforeCompile = (shader, renderer) => {
    inner.call(mat, shader, renderer);
    inject(shader, u);
  };
  mat.customProgramCacheKey = () => "herd-tone";
  return mat;
}

/** 轮廓：背面沿烘焙的 aOutDir 外扩固定像素宽，随实例一起隐藏。 */
export function createHerdOutlineMaterial(u: HerdUniforms, px = 1): MeshBasicMaterial {
  const mat = new MeshBasicMaterial({ color: INK.ink, side: BackSide, toneMapped: false });
  const own = { uOutlinePx: { value: px } };
  mat.onBeforeCompile = (shader) => {
    inject(shader, u, "attribute vec3 aOutDir;\nuniform float uOutlinePx;\n");
    Object.assign(shader.uniforms, own);
    shader.vertexShader = shader.vertexShader.replace(
      "#include <project_vertex>",
      /* glsl */ `#include <project_vertex>
      #ifdef USE_INSTANCING
      vec3 inkDir = normalize((modelViewMatrix * instanceMatrix * vec4(aOutDir, 0.0)).xyz);
      #else
      vec3 inkDir = normalize((modelViewMatrix * vec4(aOutDir, 0.0)).xyz);
      #endif
      float inkW = uOutlinePx * 2.0 * (-mvPosition.z) / (projectionMatrix[1][1] * uViewportH);
      mvPosition.xyz += inkDir * inkW * herdVisible;
      gl_Position = projectionMatrix * mvPosition;`,
    );
  };
  mat.customProgramCacheKey = () => "herd-outline";
  return mat;
}
```

注意 `inject` 先把 `#include <project_vertex>` 替换成"VERT_BODY + #include <project_vertex>"，轮廓随后的 replace 命中的是保留下来的那一处 `#include <project_vertex>`，所以外扩代码在投影之后、VERT_BODY 在投影之前。

```ts
import { Vector3, type Camera } from "three";
import { herdModelPx } from "../../../simulation/abundance";
import { lonLatToXYZ } from "../../../utils/geo";
import type { PlacedIndividual } from "./herdRuntime";

export const SURFACE_R = 1.004;
const v = new Vector3();
const n = new Vector3();
const toCam = new Vector3();

/** 个体在屏幕上的中心（客户端坐标，模型脚下向上半个尺寸）与点选半径；背面时为 null。 */
export function projectIndividual(p: PlacedIndividual, camera: Camera, rect: DOMRect, altitude: number): { x: number; y: number; r: number } | null {
  v.set(...lonLatToXYZ(p.lonLat[0], p.lonLat[1], SURFACE_R));
  n.copy(v).normalize();
  if (n.dot(toCam.copy(camera.position).sub(v).normalize()) <= 0.05) return null;
  v.project(camera);
  const px = herdModelPx(p.sizeM, altitude);
  return { x: rect.left + ((v.x + 1) / 2) * rect.width, y: rect.top + ((1 - v.y) / 2) * rect.height - px / 2, r: Math.max(8, px / 2 + 4) };
}

export function pickIndividual(x: number, y: number, camera: Camera, rect: DOMRect, altitude: number, placed: Iterable<PlacedIndividual>): PlacedIndividual | null {
  let best: PlacedIndividual | null = null;
  let bestD = Infinity;
  for (const p of placed) {
    const s = projectIndividual(p, camera, rect, altitude);
    if (!s) continue;
    const d = Math.hypot(s.x - x, s.y - y);
    if (d <= s.r && d < bestD) {
      best = p;
      bestD = d;
    }
  }
  return best;
}
```

```ts
import type { VisualizationAsset } from "../../../data/schema";
import { buildRig } from "../../Models/builders";
import { resetPose } from "../../Models/pose";
import type { Lod, Rig, Style } from "../../Models/rig";

const MAX_FREE_PER_KEY = 16;

/** 完整档的 Rig 对象池：升 / 降档频繁，避免反复构建与释放 GPU 资源。 */
export class RigPool {
  private free = new Map<string, Rig[]>();
  private disposed = false;

  private key(a: VisualizationAsset, lod: Lod, style: Style) {
    return `${a.id}|${lod}|${style}`;
  }

  acquire(a: VisualizationAsset, lod: Lod, style: Style): Rig {
    const r = this.free.get(this.key(a, lod, style))?.pop();
    if (r) {
      resetPose(r);
      return r;
    }
    return buildRig(a, lod, style);
  }

  release(r: Rig, a: VisualizationAsset, lod: Lod, style: Style): void {
    const k = this.key(a, lod, style);
    const list = this.free.get(k) ?? [];
    if (this.disposed || list.length >= MAX_FREE_PER_KEY) {
      r.dispose();
      return;
    }
    list.push(r);
    this.free.set(k, list);
  }

  get size(): number {
    let n = 0;
    for (const l of this.free.values()) n += l.length;
    return n;
  }

  dispose(): void {
    for (const l of this.free.values()) l.forEach((r) => r.dispose());
    this.free.clear();
    this.disposed = true;
  }
}
```

```tsx
 * 跑 herdSim 行为。升档个体在实例档中隐藏；出现 / 消失各有 0.25 s 缩放过渡。
 */

const PROMOTE = { max: 300, keep: 330 };
const MIN_PX = 14;
const APPEAR_S = 0.25;
const RECHECK_S = 0.5;

interface Entry {
  key: string;
  agent: HerdAgent;
  asset: VisualizationAsset;
  appear: number;
  leaving: boolean;
}

function promotionCandidates(): string[] {
  const v = useViewStore.getState();
  if (v.altitude >= CONTINENT_ALTITUDE) return [];
  const center: LonLat = [v.lon, v.lat];
  const c: { k: string; d: number }[] = [];
  for (const p of herdRuntime.placed.values()) {
    if (p.mobile && herdModelPx(p.sizeM, v.altitude) >= MIN_PX) c.push({ k: p.key, d: angularDistanceDeg(center, p.lonLat) });
  }
  return c.sort((a, b) => a.d - b.d).map((x) => x.k);
}

/** 按候选更新条目；返回是否新增了条目（需要重新渲染）。 */
function syncEntries(map: Map<string, Entry>, sorted: string[], low: boolean, rand: () => number): boolean {
  const current = new Set([...map.values()].filter((e) => !e.leaving).map((e) => e.key));
  const max = low ? PROMOTE.max / 2 : PROMOTE.max;
  const keep = low ? PROMOTE.keep / 2 : PROMOTE.keep;
  const next = choosePromoted(sorted, current, max, keep);
  for (const e of map.values()) if (!next.has(e.key)) e.leaving = true;
  let added = false;
  for (const k of next) {
    const e = map.get(k);
    if (e) {
      e.leaving = false;
      continue;
    }
    const p = herdRuntime.placed.get(k);
    const sp = p && SPECIES_BY_ID[p.speciesId];
    const asset = sp?.visualizationId ? VISUAL_BY_ID[sp.visualizationId] : undefined;
    if (!p || !sp || !asset) continue;
    const agent = createHerdAgent(k, traitsFor(sp, asset, clipsFor(asset.archetype)), p.lonLat, p.siteGroup, rand);
    map.set(k, { key: k, agent, asset, appear: 0, leaving: false });
    herdRuntime.promoted.add(k);
    added = true;
  }
  if (added) herdRuntime.version++;
  return added;
}

const easeOut = (x: number) => 1 - (1 - x) * (1 - x);

function HerdCreature({ entry, pool, lod, style, animT }: { entry: Entry; pool: RigPool; lod: Lod; style: Style; animT: RefObject<number> }) {
  const rig = useMemo(() => pool.acquire(entry.asset, lod, style), [pool, entry.asset, lod, style]);
  useEffect(() => () => pool.release(rig, entry.asset, lod, style), [pool, rig, entry.asset, lod, style]);
  const selectSpecies = useSelectionStore((s) => s.selectSpecies);
  const ref = useRef<Group>(null);
  const tmp = useMemo(() => ({ p: new Vector3(), n: new Vector3(), c: new Vector3() }), []);
  const lastClock = useRef(animT.current);
  const frame = useRef(0);

  useFrame(({ camera, size }) => {
    const g = ref.current;
    if (!g) return;
    const a = entry.agent;
    const { clip, rate } = herdClip(a.state);
    a.animT += (animT.current - lastClock.current) * rate;
    lastClock.current = animT.current;
    const ll = agentLonLat(a);
    tmp.p.set(...lonLatToXYZ(ll[0], ll[1], SURFACE_R));
    tmp.n.copy(tmp.p).normalize();
    g.visible = tmp.n.dot(tmp.c.copy(camera.position).sub(tmp.p).normalize()) > 0.05;
    if (!g.visible) return;
    g.position.copy(tmp.p);
    surfaceQuaternion(ll, a.heading, g.quaternion);
    const cam = camera as PerspectiveCamera;
    const dist = cam.position.distanceTo(tmp.p);
    const worldPerPx = (2 * dist * Math.tan((cam.fov * Math.PI) / 360)) / Math.max(1, size.height);
    const px = herdModelPx(entry.asset.sizeM, useViewStore.getState().altitude);
    g.scale.setScalar(((worldPerPx * px) / rig.extent) * easeOut(Math.max(0, Math.min(1, entry.appear))));
    frame.current++;
    if (frame.current % updateInterval(dist / (worldPerPx * px), false, useSettingsStore.getState().lowPerformance) !== 0) return;
    applyClip(rig, entry.asset.archetype, clip, a.animT, a.phase);
  });

  const onClick = (e: ThreeEvent<MouseEvent>) => {
    if (e.delta > 4) return;
    e.stopPropagation();
    selectSpecies(entry.agent.traits.speciesId);
  };

  return (
    <group ref={ref} onClick={onClick} name={`herd-agent-${entry.key}`}>
      <primitive object={rig.root} />
    </group>
  );
}

export function HerdAgents() {
  const style = useSettingsStore((s) => s.visualStyle);
  const lod: Lod = useSettingsStore((s) => s.modelDetail) === "high" ? 0 : 1;
  const pool = useMemo(() => new RigPool(), []);
  useEffect(() => () => pool.dispose(), [pool]);
  const entries = useRef(new Map<string, Entry>());
  const [keys, setKeys] = useState<string[]>([]);
  const rand = useMemo(() => seededRandom(7), []);
  const animT = useAnimationClock();
  const recheck = useRef(0);

  useEffect(
    () => () => {
      for (const k of entries.current.keys()) herdRuntime.promoted.delete(k);
      herdRuntime.version++;
      herdRuntime.agents = 0;
      herdRuntime.chases = 0;
    },
    [],
  );

  useFrame((_, dt) => {
    const s = useSettingsStore.getState();
    const map = entries.current;
    let changed = false;
    recheck.current -= dt;
    if (recheck.current <= 0) {
      recheck.current = RECHECK_S;
      changed = syncEntries(map, promotionCandidates(), s.lowPerformance, rand);
    }
    const list: HerdAgent[] = [];
    for (const [k, e] of map) {
      const p = herdRuntime.placed.get(k);
      if (!p) {
        // 物种离开生存范围或不再被散布：立即清除（追逐配对由 herdSim 自动结束）
        map.delete(k);
        herdRuntime.promoted.delete(k);
        herdRuntime.version++;
        changed = true;
        continue;
      }
      e.agent.home = p.lonLat;
      list.push(e.agent);
    }
    stepHerd(list, dt, animT.current, s.animationsEnabled && s.behaviorEnabled, rand);
    for (const [k, e] of map) {
      e.appear += (e.leaving ? -dt : dt) / APPEAR_S;
      if (e.leaving && e.appear <= 0) {
        map.delete(k);
        herdRuntime.promoted.delete(k);
        herdRuntime.version++;
        changed = true;
      } else e.appear = Math.min(1, e.appear);
    }
    herdRuntime.agents = map.size;
    herdRuntime.chases = list.filter((a) => a.state === "chase").length;
    if (changed) setKeys([...map.keys()]);
  });

  return (
    <group name="herd-agents">
      {keys.map((k) => {
        const e = entries.current.get(k);
        return e ? <HerdCreature key={k} entry={e} pool={pool} lod={lod} style={style} animT={animT} /> : null;
      })}
    </group>
  );
}
```

```ts
import { expect, test, type Page } from "@playwright/test";

interface HerdsHandle {
  stats(): { instances: number; agents: number; chases: number; species: string };
  focus(lon: number, lat: number, altitude: number): void;
  pickPoint(): { x: number; y: number; speciesId: string; name: string } | null;
}
type W = { __herds: HerdsHandle; __timeEngine: { pause(): void; seek(t: number): void } };

const stats = (page: Page) => page.evaluate(() => (window as unknown as W).__herds.stats());

async function seek(page: Page, yearsAgo: number) {
  await page.evaluate((t) => (window as unknown as W).__timeEngine.seek(t), yearsAgo);
  await expect(page.getByTestId("geo-caption")).toHaveAttribute("data-load", "ready");
}

test("herds: far-view instances, click to details, continental agents, era switch", async ({ page }) => {
  test.setTimeout(120_000);
  await page.goto("/");
  await expect(page.locator("canvas")).toBeVisible();
  await expect(page.getByTestId("geo-caption")).toHaveAttribute("data-load", "ready");
  await page.evaluate(() => (window as unknown as W).__timeEngine.pause());
  await seek(page, 68e6);

  // 远景：有实例，不超过上限
  await expect.poll(async () => (await stats(page)).instances, { timeout: 20_000 }).toBeGreaterThan(0);
  const far = await stats(page);
  expect(far.instances).toBeLessThanOrEqual(2000);

  // 点击实例打开该物种详情
  const pt = await page.evaluate(() => (window as unknown as W).__herds.pickPoint());
  expect(pt).not.toBeNull();
  await page.mouse.click(pt!.x, pt!.y);
  await expect(page.getByTestId("details-card")).toHaveAttribute("data-kind", "species");
  await page.getByTestId("details-close").click();

  // 洲视图：出现完整档，不超过 300
  await page.evaluate(() => (window as unknown as W).__herds.focus(-105, 45, 0.5));
  await expect.poll(async () => (await stats(page)).agents, { timeout: 20_000 }).toBeGreaterThan(0);
  expect((await stats(page)).agents).toBeLessThanOrEqual(300);

  // 换时代：个体集合切换
  await seek(page, 30e6);
  await expect.poll(async () => (await stats(page)).species, { timeout: 20_000 }).not.toBe(far.species);
});
```
